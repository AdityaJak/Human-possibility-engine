import sys, time
import numpy as np, torch
sys.path.insert(0, '.')
torch.set_num_threads(1)
from u1.env import make_universe
from u1.metarl import MetaRLConfig, train_metarl, MetaRLAgent
from u1.play import play
u = make_universe(0)
cfg = MetaRLConfig(updates=int(sys.argv[1]), hidden=int(sys.argv[2]), lr=float(sys.argv[3]))
t0 = time.time()
net, info = train_metarl(u, cfg, log=lambda m: print(m, flush=True))
print(info)
for g in (True, False):
    r = [play(MetaRLAgent(net, cfg.hidden, g, np.random.default_rng(s)), u, "iid", s)["norm"] for s in range(200)]
    print("greedy" if g else "sample", "iid dev 0-199", round(np.mean(r), 3))
